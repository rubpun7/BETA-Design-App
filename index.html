function handleAction(e) { if (canvas.width === 0) return; const rect = canvas.getBoundingClientRect(); const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; const x = Math.floor((clientX - rect.left) * (canvas.width / rect.width)); const y = Math.floor((clientY - rect.top) * (canvas.height / rect.height)); // ตรวจสอบสีจาก Canvas หลัก (ที่รวมทุกเลเยอร์แล้ว) เพื่อหาขอบเขตชิ้นงาน const targetColor = ctx.getImageData(x, y, 1, 1).data; const fillRGB = hexToRgb(document.getElementById('colorPicker').value); // ป้องกันการเติมทับเส้นสีดำหรือสีเข้ม (เส้นขอบชิ้นงาน) if (targetColor[0] < 80 && targetColor[1] < 80 && targetColor[2] < 80) return; saveHistory(); // ส่งค่า targetColor จากหน้าจอหลักไปเพื่อให้ Flood Fill รู้ขอบเขต floodFill(x, y, fillRGB, targetColor); } function floodFill(startX, startY, fillColor, targetColor) { // ดึงข้อมูลพิกเซลจาก Canvas หลักเพื่อเช็คขอบเขต (Boundary Check) const screenData = ctx.getImageData(0, 0, canvas.width, canvas.height).data; // ดึงข้อมูลจาก colorCanvas เพื่อเขียนสีลงไป (Writing Layer) const colorImageData = colorCtx.getImageData(0, 0, colorCanvas.width, colorCanvas.height); const colorData = colorImageData.data; if (colorsMatch(targetColor, [fillColor.r, fillColor.g, fillColor.b, 255])) return; const stack = [[startX, startY]]; const visited = new Uint8Array(canvas.width * canvas.height); while (stack.length > 0) { const [x, y] = stack.pop(); const idx = (y * canvas.width + x); const pos = idx * 4; if (x < 0 || x >= canvas.width || y < 0 || y >= canvas.height || visited[idx]) continue; // ตรวจสอบพิกเซลจากหน้าจอหลักว่าสีตรงกับจุดที่จิ้มหรือไม่ const currentColor = [screenData[pos], screenData[pos+1], screenData[pos+2], screenData[pos+3]]; if (colorsMatch(currentColor, targetColor)) { visited[idx] = 1; // เขียนสีใหม่ลงในเลเยอร์สี (colorCanvas) setPixel(x, y, fillColor, colorData); stack.push([x + 1, y], [x - 1, y], [x, y + 1], [x, y - 1]); } } colorCtx.putImageData(colorImageData, 0, 0); redrawAllLayers(); // สั่งวาดใหม่เพื่อรวมสีเข้ากับเส้นขอบ }
